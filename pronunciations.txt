# Файл перечитывается сам, перезапуск бота не нужен.
#
# Примеры:
Аура = +Аура
MirFeeD = Мирф+ид
Кактусик = Как+тусик
Пашка = П+ашка
кринж = кр+инж
агент = аг+ент
скин = ск+ин
катка = катк+а
Elyve = эл+айв
Elyvka = эл+ивка

хотя = хот+я
